# Clone the repo
git clone <repository-url>

# Move into the project directory
cd <repository-name>

# Install dependencies with Bun
bun install

# Update Package.json

"scripts": {
    "dev": "bun --bun next dev",
    "build": "bun --bun next build",
    "start": "bun --bun next start",
    "lint": "eslint"
  },


# Run the dev server
bun dev
